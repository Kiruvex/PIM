    65→    65→class HazardModel:
    66→    66→    def __init__(self):
    67→    67→        self.present = {}
    68→    68→        self.wait = {}
    69→    69→        self.gaps_sorted = []
    70→    70→        self.lam = 1.0 / 36.0
    71→    71→
    72→    72→    def observe_gap(self, d):
    73→    73→        self.present[d] = self.present.get(d, 0) + 1
    74→    74→        for k in range(1, d + 1):
    75→    75→            self.wait[k] = self.wait.get(k, 0) + 1
    76→    76→        insort(self.gaps_sorted, d)
    77→    77→
    78→    78→    def refit_tail(self):
    79→    79→        gs = self.gaps_sorted
    80→    80→        n = len(gs)
    81→    81→        if n < 8:
    82→    82→            med = gs[n // 2] if gs else 18
    83→    83→            self.lam = max(1e-6, 1.0 / (2 * med))
    84→    84→            return
    85→    85→        pts = []
    86→    86→        mx = min(gs[-1], D_MAX_OBS)
    87→    87→        for d in range(TAIL_START, mx + 1, 5):
    88→    88→            n_gt = n - bisect_right(gs, d)
    89→    89→            if n_gt >= 5:
    90→    90→                pts.append((d, n_gt / n))
    91→    91→        if len(pts) < 3:
    92→    92→            self.lam = max(1e-6, 1.0 / (2 * gs[n // 2]))
    93→    93→            return
    94→    94→        xs = [p[0] for p in pts]
    95→    95→        ys = [math.log(p[1]) for p in pts]
    96→    96→        xb = sum(xs) / len(xs)
    97→    97→        yb = sum(ys) / len(ys)
    98→    98→        sxx = sum((x - xb) ** 2 for x in xs)
    99→    99→        if sxx <= 0:
   100→   100→            self.lam = max(1e-6, 1.0 / (2 * gs[n // 2]))
   101→   101→            return
   102→   102→        slope = sum((x - xb) * (y - yb) for x, y in pts) / sxx
   103→   103→        self.lam = max(1e-6, -slope)
   104→   104→
   105→   105→    def hazard(self, k):
   106→   106→        if k <= D_MAX_OBS and self.wait.get(k, 0) > 0:
   107→   107→            return (self.present.get(k, 0) + ALPHA) / (self.wait[k] + 2 * ALPHA)
   108→   108→        return min(0.99, 1.0 - math.exp(-self.lam))
   109→   109→
   110→   110→    def cost(self, k):
   111→   111→        if k <= 0:
   112→   112→            return 0.0
   113→   113→        return -math.log2(max(1.0 - self.hazard(k), 1e-12))
